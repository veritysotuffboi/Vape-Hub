local ENEMY_MODES = {"Enemy team","Cops (Guards)","Criminals","Everyone"}
local enemyMode = 1
local function isEnemy(p)
    if not p or p == LocalPlayer then return false end
    local m = ENEMY_MODES[enemyMode]; local tn = p.Team and p.Team.Name:lower() or ""
    if m == "Everyone" then return true
    elseif m == "Cops (Guards)" then return (tn:find("guard") or tn:find("police") or tn:find("cop")) ~= nil
    elseif m == "Criminals" then return (tn:find("crim") or tn:find("mafia")) ~= nil
    else return p.Team ~= LocalPlayer.Team end
end

local fakeFloor = nil

local function acquireFloor() fakeFloorUsers+=1; spawnFakeFloor() end

local _noclipTemp = false
local _noclipTempConns = {}
local function tempNoclipOn()
    if _noclipTemp then return end; _noclipTemp=true
    local c=char(); if not c then return end
    for _,v in ipairs(c:GetDescendants()) do
        if v:IsA("BasePart") then
            pcall(function() v.CanCollide=false end)
            local conn=v:GetPropertyChangedSignal("CanCollide"):Connect(function()
                if _noclipTemp and v.CanCollide then v.CanCollide=false end end)
            _noclipTempConns[#_noclipTempConns+1]=conn
        end
    end
end
local function tempNoclipOff()
    if not _noclipTemp then return end; _noclipTemp=false
    for _,c in ipairs(_noclipTempConns) do pcall(function() c:Disconnect() end) end
    _noclipTempConns={}
    if not F.Mov.Noclip then
        local c=char()
        if c then for _,v in ipairs(c:GetDescendants()) do
            if v:IsA("BasePart") and v.Name~="HumanoidRootPart" then pcall(function() v.CanCollide=true end) end
        end end
    end
end

local function tweenTP(cf, dur)
    local r=root(); if not (r and cf) then return end; dur=dur or 0.22
    tempNoclipOn(); acquireFloor()
    local startCF=r.CFrame; local t0=os.clock()
    while true do
        local a=(os.clock()-t0)/dur; if a>=1 then break end
        a=1-(1-a)*(1-a)
        pcall(function() r.CFrame=startCF:Lerp(cf,a); r.AssemblyLinearVelocity=Vector3.zero end)
        RunService.Heartbeat:Wait()
    end
    pcall(function() r.CFrame=cf; r.AssemblyLinearVelocity=Vector3.zero end)
    RunService.Heartbeat:Wait(); releaseFloor(); tempNoclipOff()
end
local function tpMove(cf) if cf then task.spawn(tweenTP,cf,0.22) end end
local function tpInstant(cf) local r=root(); if r and cf then r.CFrame=cf end end

local carFlyConn = nil
local carFlyObjects = {}
local function GetSeatVehicle()
    local c=char(); if not c then return nil end
    local h=c:FindFirstChildOfClass("Humanoid"); if not h then return nil end
    local seat=h.SeatPart; if not seat then return nil end
    return seat:FindFirstAncestorWhichIsA("Model")
end
local function stopCarFly()
    if carFlyConn then pcall(function() carFlyConn:Disconnect() end); carFlyConn=nil end
    for _,obj in ipairs(carFlyObjects) do pcall(function() obj:Destroy() end) end; carFlyObjects={}
end
local function setupCarFly()
    stopCarFly(); if not F.Mov.CarFly then return end
    local vehicle=GetSeatVehicle(); if not vehicle then return end
    local pp=vehicle.PrimaryPart
    if not pp then
        for _,v in ipairs(vehicle:GetDescendants()) do
            if v:IsA("BasePart") and (v.Name=="Body" or v.Name=="Chassis" or v.Name=="DriveSeat" or v:IsA("VehicleSeat")) then pp=v; break end
        end
    end
    if not pp then return end
    for _,v in ipairs(vehicle:GetDescendants()) do if v:IsA("BasePart") then v.Anchored=false end end
    local bv=Instance.new("BodyVelocity"); bv.MaxForce=Vector3.new(1e9,1e9,1e9); bv.Velocity=Vector3.zero; bv.Parent=pp; table.insert(carFlyObjects,bv)
    local bg=Instance.new("BodyGyro"); bg.MaxTorque=Vector3.new(1e9,1e9,1e9); bg.D=500; bg.P=1e5; bg.CFrame=pp.CFrame; bg.Parent=pp; table.insert(carFlyObjects,bg)
    for _,v in ipairs(vehicle:GetDescendants()) do
        if v:IsA("BasePart") then
            local nc=v:GetPropertyChangedSignal("CanCollide"):Connect(function() v.CanCollide=false end)
            table.insert(carFlyObjects,nc)
        end
    end
    carFlyConn=RunService.Heartbeat:Connect(function(dt)
        if not F.Mov.CarFly or not pp or not pp.Parent then stopCarFly(); return end
        local c2=cam().CFrame
        local md=Vector3.zero
        if key(Enum.KeyCode.W) then md+=c2.LookVector end
        if key(Enum.KeyCode.S) then md-=c2.LookVector end
        if key(Enum.KeyCode.A) then md-=c2.RightVector end
        if key(Enum.KeyCode.D) then md+=c2.RightVector end
        if key(Enum.KeyCode.Space) then md+=Vector3.new(0,1,0) end
        if key(Enum.KeyCode.LeftControl) then md-=Vector3.new(0,1,0) end
        if md.Magnitude>0 then bv.Velocity=md.Unit*F.Mov.CarFlySpeed else bv.Velocity=Vector3.zero end
        local _,yaw,_=c2:ToEulerAnglesYXZ()
        bg.CFrame=CFrame.new(pp.Position)*CFrame.Angles(0,yaw,0)
    end)
end

local Aim = {}
do
    local function getTarget(fovOverride, triggerMode)
        local fov=fovOverride or F.Aim.FOV
        local mousePos=aimOrigin()
        local best,bestDist=nil,math.huge
        for _,p in ipairs(Players:GetPlayers()) do
            if p~=LocalPlayer and isEnemy(p) then
                local ok,c,h=alive(p); if not ok then continue end
                local part=c:FindFirstChild(F.Aim.Part) or c:FindFirstChild("HumanoidRootPart")
                if part then
                    local sp,vis=cam():WorldToViewportPoint(part.Position)
                    if vis then
                        local d=(Vector2.new(sp.X,sp.Y)-mousePos).Magnitude
                        if d<fov and d<bestDist then
                            if F.Aim.WallCheck then
                                local rp=RaycastParams.new()
                                rp.FilterDescendantsInstances={char(),c}
                                rp.FilterType=Enum.RaycastFilterType.Exclude
                                local o=cam().CFrame.Position
                                local ray=Workspace:Raycast(o,part.Position-o,rp)
                                if ray and not ray.Instance:IsDescendantOf(c) then continue end
                            end
                            best=part; bestDist=d
                        end
                    end
                end
            end
        end
        return best
    end
    Aim.getTarget=getTarget
end

doRage = function(target)
    if not target then return end
    task.spawn(function()
        Hub._raging=function() return true end
        local guns={"AK-47","MP5","M4A1","Remington 870","M9","Revolver"}
        local gun
        for _,g in ipairs(guns) do
            gun=GetTool(g)
            if gun then break end
            GetGun(g); task.wait(0.4)
            gun=GetTool(g); if gun then break end
        end
        local tr=target.Character and (target.Character:FindFirstChild("HumanoidRootPart") or target.Character:FindFirstChild("Torso"))
        if tr then tpInstant(tr.CFrame*CFrame.new(0,0,-2.5)) end
        task.wait(0.1)
        local h=target.Character and target.Character:FindFirstChildOfClass("Humanoid")
        if h and h.Health>0 then fireClick() end
        task.wait(3)
        Hub._raging=function() return false end
    end)
end

local function isAlive()
    local h=hum(); return h and h.Health>0
end

local function getEntityList()
    local list={}
    for _,p in ipairs(Players:GetPlayers()) do
        if p~=LocalPlayer then
            local ok,c,h=alive(p)
            if ok and c then
                local rp=c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("Torso")
                local head=c:FindFirstChild("Head")
                if rp and h then
                    list[#list+1]={
                        Player=p, Character=c, Humanoid=h, Health=h.Health,
                        RootPart=rp, Head=head or rp,
                        Targetable=isEnemy(p)
                    }
                end
            end
        end
    end
    return list
end

local function getClosestEntity(range, part, attackCheck)
    local entities=getEntityList()
    local r=getRoot(); if not r then return nil end
    local best,bestDist=nil,math.huge
    for _,ent in ipairs(entities) do
        if attackCheck and not ent.Targetable then continue end
        local ep=ent[part or "RootPart"]
        if ep then
            local d=(ep.Position-r.Position).Magnitude
            if d<(range or 12) and d<bestDist then
                best=ent; bestDist=d
            end
        end
    end
    return best
end

local pl = {}
local Spring = {}
do
    Spring.__index = Spring
    function Spring.new(props)
        local p = props or {}
        return setmetatable({
            Target=Vector3.new(), Position=Vector3.new(), Velocity=Vector3.new(),
            Mass=p.Mass or 5, Force=p.Force or 50, Damping=p.Damping or 4, Speed=p.Speed or 4
        }, Spring)
    end
    function Spring:Update(dt)
        local iters = dt / ((1/60)/8)
        local sdt = dt * self.Speed / iters
        for _=1, math.round(iters) do
            local acc = (self.Target - self.Position) * self.Force / self.Mass
            acc -= self.Velocity * self.Damping
            self.Velocity += acc * sdt
            self.Position += self.Velocity * sdt
        end
        return self.Position
    end
end

                    pcall(restorefunction, pl.GunTracers.createTaser)
                    pcall(restorefunction, pl.GunTracers.createSniper)

                    pcall(hookfunction, pl.GunTracers.createTaser,  oldTaser)
                    pcall(hookfunction, pl.GunTracers.createSniper, oldSniper)
                end
                oldBullet=nil; oldTaser=nil; oldSniper=nil
            end
        end
    end

local antiInvisThreads = {}
local antiInvisConns = {}

local function antiInvisAnimAdded(anim, plr)
    if not antiInvisWhitelist[anim.Animation.AnimationId] and plr then
        if antiInvisThreads[anim] then task.cancel(antiInvisThreads[anim]) end
        antiInvisThreads[anim] = task.spawn(function()
            repeat
                pcall(function() anim:AdjustWeight(0, 0) end)
                task.wait()
            until not (anim.IsPlaying and F.VapeM.AntiInvisible)
            antiInvisThreads[anim] = nil
        end)
    end
end

local function hookAntiInvisPlayer(p)
    local c = p.Character
    if not c then return end
    local animator = c:FindFirstChildOfClass("Animator") or c:WaitForChild("Animator", 3)
    if not animator then return end
    local ac = animator.AnimationPlayed:Connect(function(anim) antiInvisAnimAdded(anim, p) end)
    antiInvisConns[#antiInvisConns+1] = ac
    for _,anim in ipairs(animator:GetPlayingAnimationTracks()) do
        task.spawn(antiInvisAnimAdded, anim, p)
    end
end

local function startAntiInvisible()
    for _,p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then task.spawn(hookAntiInvisPlayer, p) end
    end
    antiInvisConns[#antiInvisConns+1] = Players.PlayerAdded:Connect(function(p)
        p.CharacterAdded:Connect(function()
            if not F.VapeM.AntiInvisible then return end
            task.spawn(hookAntiInvisPlayer, p)
        end)
    end)
    for _,p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            antiInvisConns[#antiInvisConns+1] = p.CharacterAdded:Connect(function()
                if not F.VapeM.AntiInvisible then return end
                task.spawn(hookAntiInvisPlayer, p)
            end)
        end
    end
end

local function stopAntiInvisible()
    for _,c in ipairs(antiInvisConns) do pcall(function() c:Disconnect() end) end
    antiInvisConns = {}
    for _,t in pairs(antiInvisThreads) do pcall(function() task.cancel(t) end) end
    antiInvisThreads = {}
end

local silentAimOld = nil
local function startSilentAim()
    if not (pl.Bullet and hookfunction) then return end
    if silentAimOld then return end
    local rand = Random.new()
    silentAimOld = hookfunction(pl.Bullet, function(origin, direction, ...)
        if rand:NextNumber(0,100) > F.VapeM.SilentAimHitChance then return silentAimOld(origin, direction, ...) end
        local targetPart = rand:NextNumber(0,100) < F.VapeM.SilentAimHeadshotChance and "Head" or "HumanoidRootPart"
        local best, bestDist = nil, math.huge
        local r = root()
        if not r then return silentAimOld(origin, direction, ...) end
        for _,p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and isEnemy(p) then
                local ok,c,h = alive(p); if not ok then continue end
                local part = c:FindFirstChild(targetPart) or c:FindFirstChild("HumanoidRootPart")
                if part then
                    local d = (part.Position - r.Position).Magnitude
                    if d <= F.VapeM.SilentAimRange and d < bestDist then
                        if not F.VapeM.SilentAimWallbang then
                            local rp = RaycastParams.new()
                            rp.FilterDescendantsInstances = {char(), c}
                            rp.FilterType = Enum.RaycastFilterType.Exclude
                            local ray = Workspace:Raycast(origin, part.Position - origin, rp)
                            if ray and not ray.Instance:IsDescendantOf(c) then continue end
                        end
                        best = part; bestDist = d
                    end
                end
            end
        end
        if best then
            return silentAimOld(origin, best.Position, ...)
        end
        return silentAimOld(origin, direction, ...)
    end)
end

local function stopSilentAim()
    if silentAimOld and pl.Bullet and hookfunction then
        pcall(function()
            if restorefunction then restorefunction(pl.Bullet)
            else hookfunction(pl.Bullet, silentAimOld) end
        end)
        silentAimOld = nil
    end
end

local diIndi, diIndiPart, diIndiThread = nil, nil, nil
local diTent, diTtimer, diThealth = nil, 0, 0

local function diCreateIndicator(damage, pos)
    if diIndiThread then
        task.cancel(diIndiThread); diIndiThread=nil
        if diIndi then diIndi.Text = tostring(math.ceil(tonumber(diIndi.Text or "0") + damage)) end
        if diIndiPart then diIndiPart.Position = pos end
    else
        diIndiPart = Instance.new("Part")
        diIndiPart.Size = Vector3.zero; diIndiPart.Position = pos
        diIndiPart.CanCollide=false; diIndiPart.CanQuery=false
        diIndiPart.Anchored=true; diIndiPart.Parent=Workspace
        local bb = Instance.new("BillboardGui")
        bb.Adornee=diIndiPart; bb.Size=UDim2.new(15,250,15,250)
        bb.AlwaysOnTop=true; bb.Parent=diIndiPart
        diIndi = Instance.new("TextLabel")
        diIndi.BackgroundTransparency=1; diIndi.TextStrokeTransparency=0
        diIndi.Size=UDim2.fromScale(1,0.075)
        diIndi.Position=UDim2.fromScale(0.5,0.5)
        diIndi.AnchorPoint=Vector2.new(0.5,0.5)
        diIndi.Text=tostring(math.ceil(damage))
        diIndi.TextColor3=Color3.fromRGB(255,220,60)
        diIndi.TextScaled=true; diIndi.Font=Enum.Font.GothamBlack
        diIndi.Parent=bb
    end
    diIndiThread = task.spawn(function()
        local sign = math.sign(math.random()-0.5)
        local t0 = os.clock()
        while true do
            local pct = math.clamp((os.clock()-t0)/0.6, 0, 1)
            if diIndi then
                diIndi.Position = UDim2.new(0.5 + 0.125*pct*sign, 0, 0.5 - 0.15*pct, 0)
                diIndi.TextTransparency = pct
            end
            if pct >= 1 then break end
            RunService.RenderStepped:Wait()
        end
        if diIndiPart then pcall(function() diIndiPart:Destroy() end) end
        diIndiPart=nil; diIndi=nil; diIndiThread=nil
    end)
end

local function startDamageIndicator()
    TracerHook:Add("DamageIndicator", function(...)
        local ok, part = pcall(debug.getstack, 4, 17)
        if ok and typeof(part)=="Instance" then
            for _,ent in ipairs(getEntityList()) do
                if part:IsDescendantOf(ent.Character) then
                    if diTtimer <= os.clock() or ent.Player ~= diTent then
                        diThealth = ent.Health
                    end
                    diTent = ent.Player; diTtimer = os.clock() + 0.5
                    break
                end
            end
        end
        return false
    end, 0)
    vapeClean(RunService.Heartbeat:Connect(function()
        if not (diTent and diTtimer > os.clock()) then return end
        for _,ent in ipairs(getEntityList()) do
            if ent.Player == diTent and diThealth > ent.Health then
                local dmg = diThealth - ent.Health
                local head = ent.Head
                if head then diCreateIndicator(dmg, head.Position + Vector3.new(0,2,0)) end
                diThealth = ent.Health
            end
        end
    end))
end

local function stopDamageIndicator()
    TracerHook:Remove("DamageIndicator")
    if diIndiPart then pcall(function() diIndiPart:Destroy() end); diIndiPart=nil; diIndi=nil end
    if diIndiThread then task.cancel(diIndiThread); diIndiThread=nil end
end

local vmMoveSpring = Spring.new()
local vmAimSpring  = Spring.new({Speed=15})
local vmOldTool, vmClone, vmHandle = nil, nil, nil
local vmShootTimer = os.clock()
local vmAimTimer   = os.clock()
local vmAimVec     = Vector3.new()
local vmConn       = nil

local function vmCleanup()
    if vmOldTool then
        for _,v in ipairs(vmOldTool:GetDescendants()) do
            if v:IsA("BasePart") or v:IsA("Texture") or v:IsA("Decal") then
                pcall(function() v.LocalTransparencyModifier = 0 end)
            end
        end
        vmOldTool = nil
    end
    if vmClone then pcall(function() vmClone:Destroy() end); vmClone=nil; vmHandle=nil end
end

local function vmToolAdded(tool)
    if not (tool and tool:IsA("Tool")) then return end
    vmCleanup()
    vmOldTool = tool
    vmClone = tool:Clone()
    vmHandle = vmClone:FindFirstChild("Handle")
    vmClone.Parent = cam()
    for _,v in ipairs(vmClone:GetDescendants()) do
        if v:IsA("BasePart") then v.CanCollide=false; v.Anchored=true end
    end
    for _,v in ipairs(tool:GetDescendants()) do
        if v:IsA("BasePart") or v:IsA("Texture") or v:IsA("Decal") then
            pcall(function() v.LocalTransparencyModifier = 1 end)
        end
    end
end

local function startViewmodel()
    local c = char()
    if c then
        local t = c:FindFirstChildWhichIsA("Tool")
        if t then vmToolAdded(t) end
        vapeClean(c.ChildAdded:Connect(vmToolAdded))
        vapeClean(c.ChildRemoved:Connect(function(tool)
            if tool == vmOldTool then vmCleanup() end
        end))
    end
    vapeClean(Players.LocalPlayer.CharacterAdded:Connect(function(nc)
        vmCleanup()
        task.wait(0.5)
        local t2 = nc:FindFirstChildWhichIsA("Tool")
        if t2 then vmToolAdded(t2) end
        vapeClean(nc.ChildAdded:Connect(vmToolAdded))
        vapeClean(nc.ChildRemoved:Connect(function(tool)
            if tool == vmOldTool then vmCleanup() end
        end))
    end))
    TracerHook:Add("Viewmodel", function(...)
        vmShootTimer = os.clock() + 0.3
        return false
    end, 0)
    vmConn = vapeClean(RunService.RenderStepped:Connect(function(dt)
        if not (vmHandle and vmHandle.Parent) then return end
        local c2 = cam().CFrame
        local r2 = root()
        vmMoveSpring.Target = r2 and r2.AssemblyLinearVelocity * 0.005 or Vector3.new()
        local cf = c2 * CFrame.new(0.4, -0.35, -1.2) + vmMoveSpring:Update(dt)
        vmAimSpring.Target = vmAimTimer > os.clock() and CFrame.lookAt(cf.Position, vmAimVec).LookVector or c2.LookVector
        local kick = math.max(vmShootTimer - os.clock(), 0)
        vmHandle.CFrame = CFrame.lookAlong(cf.Position, vmAimSpring:Update(dt))
            * CFrame.Angles(math.rad(kick * 10), 0, 0)
            * CFrame.new(0, 0, kick)
        vmHandle.AssemblyLinearVelocity = Vector3.zero
    end))
end

local function stopViewmodel()
    TracerHook:Remove("Viewmodel")
    vmCleanup()
end

local hotbarSortList = {
    ["AK-47"]=1,["MP5"]=1,["M4A1"]=1,

local function doHotbarSort()
    if not (pl.SwitchTable and pl.SwitchUpdate) then return end
    pcall(function()
        table.sort(pl.SwitchTable, function(a,b)
            local pa = (hotbarSortList[a.Tool and a.Tool.Name] or 999)
            local pb = (hotbarSortList[b.Tool and b.Tool.Name] or 999)
            if pa ~= pb then return pa < pb end
            return (a.Slot or 0) < (b.Slot or 0)
        end)
        task.spawn(pl.SwitchUpdate)
    end)
end

        hotbarConn = vapeClean(bp.ChildAdded:Connect(function(tool)

        end))
    end
end

    if hotbarConn then pcall(function() hotbarConn:Disconnect() end); hotbarConn=nil end
end

local function findMeleeRemote()
    local candidates = {
        function() return RS:FindFirstChild("meleeEvent") end,
        function() return RS.Remotes and RS.Remotes:FindFirstChild("MeleeHit") end,
        function() return RS.Remotes and RS.Remotes:FindFirstChild("Melee") end,
        function() for _,v in ipairs(RS:GetDescendants()) do if v:IsA("RemoteEvent") and v.Name:lower():find("melee") then return v end end end,
    }
    for _,fn in ipairs(candidates) do local ok,r = pcall(fn); if ok and r then return r end end

end

local killauraThread=nil
local function startKillaura()
    killauraThread=vapeSpawn(function()
        while F.VapeM.Killaura do
            if isAlive() then
                local r=getRoot()
                local selfpos=r and r.Position
                if selfpos then
                    local entities=getEntityList()
                    local attacked=0
                    local facing=r.CFrame.LookVector*Vector3.new(1,0,1)
                    local meleeRemote = findMeleeRemote()
                    for _,ent in ipairs(entities) do
                        if not ent.Targetable then continue end
                        local ep=ent.RootPart; if not ep then continue end
                        local dist=(ep.Position-selfpos).Magnitude
                        if dist>F.VapeM.KillauraRange then continue end
                        local delta=(ep.Position-selfpos)
                        local flatDelta = delta * Vector3.new(1,0,1)
                        local angle = flatDelta.Magnitude > 0
                            and math.acos(math.clamp(facing:Dot(flatDelta.Unit),-1,1)) or 0
                        if angle>(math.rad(F.VapeM.KillauraAngle)/2) then continue end
                        if attacked>=F.VapeM.KillauraMax then break end
                        if meleeRemote then pcall(function() meleeRemote:FireServer(ent.Player,1,1) end) end
                        attacked+=1
                    end
                end
            end
            task.wait(0.05)
        end
    end)
end

local BONES={{" Head","Torso"},{"Torso","Left Arm"},{"Torso","Right Arm"},{"Torso","Left Leg"},{"Torso","Right Leg"}}
local ESP={}
do
    local hasDraw=false
    if Drawing and Drawing.new then
        local ok,probe=pcall(Drawing.new,"Square")
        if ok and probe then hasDraw=true; pcall(function() probe:Remove() end) end
    end
    local host=CoreGui
    if gethui then local ok,h=pcall(gethui); if ok and h then host=h end end
    host=host or LocalPlayer:FindFirstChildOfClass("PlayerGui")
    local pool={}
    local function nd(k,pr) local d=Drawing.new(k); for a,b in pairs(pr) do d[a]=b end; return d end
    local function build(p)
        if hasDraw then
            local ok=pcall(function()
                local bones={}; for i=1,#BONES do bones[i]=nd("Line",{Thickness=1,Color=Color3.new(1,1,1)}) end
                pool[p]={box=nd("Square",{Thickness=1,Filled=false,Color=Color3.new(1,1,1)}),
                    boxo=nd("Square",{Thickness=3,Filled=false,Color=Color3.new(0,0,0)}),
                    name=nd("Text",{Size=13,Center=true,Outline=true,Color=Color3.new(1,1,1)}),
                    dist=nd("Text",{Size=12,Center=true,Outline=true,Color=Color3.fromRGB(190,190,190)}),
                    trac=nd("Line",{Thickness=1,Color=Color3.fromRGB(255,95,95)}),
                    hpbg=nd("Square",{Thickness=1,Filled=true,Color=Color3.fromRGB(15,15,15)}),
                    hp=nd("Square",{Thickness=1,Filled=true,Color=Color3.fromRGB(90,230,150)}),bones=bones}
            end)
            if ok and pool[p] then return end; hasDraw=false; pool[p]=nil
        end
        local ok2,hl=pcall(Instance.new,"Highlight")
        if ok2 and hl then
            hl.FillTransparency=0.55; hl.OutlineColor=Color3.fromRGB(255,95,95)
            hl.FillColor=Color3.fromRGB(255,95,95); hl.Adornee=p.Character
            pcall(function() hl.Parent=host end); pool[p]={hl=hl}
        end
    end
    local function hide(o)
        if o.hl then o.hl.Adornee=nil; return end
        o.box.Visible=false; o.boxo.Visible=false; o.name.Visible=false; o.dist.Visible=false
        o.trac.Visible=false; o.hpbg.Visible=false; o.hp.Visible=false
        for _,ln in ipairs(o.bones) do ln.Visible=false end
    end
    function ESP.clear(p) local o=pool[p]; if not o then return end
        if o.hl then o.hl:Destroy() else
            for _,d in ipairs({o.box,o.boxo,o.name,o.dist,o.trac,o.hpbg,o.hp}) do pcall(function() d:Remove() end) end
            for _,ln in ipairs(o.bones) do pcall(function() ln:Remove() end) end
        end; pool[p]=nil end
    function ESP.render()
        if not attached then for _,p in ipairs(Players:GetPlayers()) do local o=pool[p]; if o then hide(o) end end; return end
        for _,p in ipairs(Players:GetPlayers()) do if p~=LocalPlayer then
            if not pool[p] then build(p) end
            local o=pool[p]; local ok,c,h=alive(p)
            local hrp=ok and (c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("Torso"))
            if not (F.ESP.On and hrp) then if o then hide(o) end
            elseif o.hl then o.hl.Adornee=c
            else
                local enemy=isEnemy(p)
                if F.ESP.EnemiesOnly and not enemy then hide(o); continue end
                local col=enemy and Color3.fromRGB(255,95,95) or Color3.fromRGB(90,230,150)
                local head=c:FindFirstChild("Head")
                local top=cam():WorldToViewportPoint((head and head.Position or hrp.Position)+Vector3.new(0,0.6,0))
                local bot=cam():WorldToViewportPoint(hrp.Position-Vector3.new(0,3.2,0))
                if top.Z>0 then
                    local h2=math.abs(top.Y-bot.Y); local w=h2*0.62; local x,y=top.X-w/2,top.Y
                    o.box.Visible=F.ESP.Box; o.boxo.Visible=F.ESP.Box; o.box.Filled=F.ESP.Filled; o.box.Transparency=F.ESP.Filled and 0.35 or 1
                    o.box.Size=Vector2.new(w,h2); o.box.Position=Vector2.new(x,y); o.boxo.Size=o.box.Size; o.boxo.Position=o.box.Position; o.box.Color=col

function Hub._unload()
    attached=false
    for _,c in ipairs(conns) do pcall(function() c:Disconnect() end) end
    clearVapeConns(); killVapeThreads()
    stopCarFly(); stopSpin(); stopAntiFling(); stopAntiKillPlane()
    stopAntiRiotShield(); stopNoJumpCooldown(); stopCameraPhase()
    stopC4ESP(); stopDisabler(); stopAntiInvisible()
    stopSilentAim(); stopBulletTracers(); stopDamageIndicator()
    stopHitSound(); stopKillSound(); stopViewmodel(); stopAutoHotbar()
    pcall(clearNoclip); pcall(ESP.destroy); pcall(stopSpectate)
    if fovCircle then pcall(function() fovCircle:Remove() end) end
    pcall(function() fullbright(false) end)
    pcall(function() cam().FieldOfView=70 end)
    pcall(function() gui:Destroy() end)
    local hh=hum(); if hh then hh.WalkSpeed=16; hh.JumpPower=50 end
    if getgenv then getgenv().vapehub=nil end
end
